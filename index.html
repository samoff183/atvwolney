const campoBusca = document.getElementById('busca'); const livros = document.querySelectorAll('.livro'); const form = document.getElementById('form-sugestao'); const campoTitulo = document.getElementById('titulo'); const mensagem = document.getElementById('mensagem'); campoBusca.addEventListener('input', () => { const termo = campoBusca.value.toLowerCase(); livros.forEach(livro => { const titulo = livro.textContent.toLowerCase(); livro.parentElement.hidden = !titulo.includes(termo); }); }); form.addEventListener('submit', (evento) => { evento.preventDefault(); const titulo = campoTitulo.value.trim(); if (titulo.length < 2) { mensagem.textContent = 'Erro! O título deve ter pelo menos 2 letras.'; } else { mensagem.textContent = 'Sugestão enviada com sucesso.'; form.reset(); } });
